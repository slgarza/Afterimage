"""Root-authorized bounded GET-only observer; no mutation or recovery invocation."""

import fcntl
import hashlib
import json
import os
import subprocess
import time
from datetime import UTC, datetime
from pathlib import Path

RUN = Path(__file__).resolve().parent
HEAD = "5ef03bcc2ae84cf25e67b83a8d5f1c9b51d1f208"
REPO = "slgarza/Afterimage"
MAX_ATTEMPTS = 100
INTERVAL_SECONDS = 60
OUT = RUN / "b096-delegated-get-observer"


def utc():
    return datetime.now(UTC).isoformat()


def write_once(path, raw):
    with os.fdopen(
        os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600), "wb"
    ) as stream:
        stream.write(raw)
        stream.flush()
        os.fsync(stream.fileno())


def encode(value):
    return (json.dumps(value, indent=2) + "\n").encode()


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def classify(payload):
    rows = payload.get("check_runs")
    total = payload.get("total_count")
    if (
        type(total) is not int
        or not isinstance(rows, list)
        or len(rows) != total
        or total > 100
    ):
        return "incomplete_or_invalid", 0
    if any(not isinstance(row, dict) for row in rows):
        return "incomplete_or_invalid", 0
    ids = [row.get("id") for row in rows]
    if any(type(value) is not int for value in ids) or len(set(ids)) != len(ids):
        return "incomplete_or_invalid", 0
    actions = [
        row for row in rows if row.get("app", {}).get("slug") == "github-actions"
    ]
    if not actions or "checks" not in {row.get("name") for row in actions}:
        return "missing_required_checks", len(actions)
    if any(row.get("head_sha") != HEAD for row in actions):
        return "wrong_head", len(actions)
    if any(
        row.get("status") == "completed" and row.get("conclusion") != "success"
        for row in actions
    ):
        return "failed_or_cancelled", len(actions)
    if all(
        row.get("status") == "completed" and row.get("conclusion") == "success"
        for row in actions
    ):
        return "all_exact_head_checks_green", len(actions)
    return "pending", len(actions)


def main():
    OUT.mkdir(mode=0o700, exist_ok=True)
    cutoff = datetime.fromisoformat(
        json.loads((RUN / "workload-release.json").read_text())["delivery_cutoff_utc"]
    )
    if cutoff.utcoffset() is None:
        raise ValueError("aware frozen cutoff required")
    lock_path = OUT / "observer.lock"
    with lock_path.open("a") as lock:
        lock_path.chmod(0o600)
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        config = {
            "kind": "root_authorized_bounded_get_observer",
            "head_sha": HEAD,
            "repository": REPO,
            "maximum_GET_attempts_in_this_observer_scope": MAX_ATTEMPTS,
            "interval_seconds": INTERVAL_SECONDS,
            "cutoff_utc": cutoff.isoformat(),
            "actual_started_at_utc": utc(),
            "other_pipeline_GETs_are_separate_and_not_claimed_in_this_counter": True,
            "mutations_or_paid_resources_authorized_by_this_observer": False,
        }
        configuration = OUT / "configuration.json"
        if not configuration.exists():
            write_once(configuration, encode(config))
        else:
            existing = json.loads(configuration.read_text())
            if any(
                existing[key] != config[key]
                for key in [
                    "head_sha",
                    "repository",
                    "maximum_GET_attempts_in_this_observer_scope",
                    "cutoff_utc",
                ]
            ):
                raise ValueError("observer scope changed")
        reservations = sorted(OUT.glob("attempt-*-reserved.json"))
        used = len(reservations)
        if used > MAX_ATTEMPTS:
            raise ValueError("invalid durable attempt count")
        print(
            json.dumps(
                {
                    "kind": "bounded_get_observer_started",
                    "at_utc": utc(),
                    "previous_attempts": used,
                    "maximum_attempts": MAX_ATTEMPTS,
                    "cutoff_utc": cutoff.isoformat(),
                }
            ),
            flush=True,
        )
        while used < MAX_ATTEMPTS and datetime.now(UTC) < cutoff:
            if (OUT / "stop-observing").exists():
                break
            began = datetime.now(UTC)
            used += 1
            prefix = OUT / f"attempt-{used:03}"
            write_once(
                prefix.with_name(prefix.name + "-reserved.json"),
                encode(
                    {
                        "attempt": used,
                        "reserved_at_utc": began.isoformat(),
                        "method": "GET",
                        "head_sha": HEAD,
                    }
                ),
            )
            remaining = (cutoff - datetime.now(UTC)).total_seconds()
            if remaining <= 0:
                break
            started = time.monotonic()
            try:
                process = subprocess.run(
                    [
                        "gh",
                        "api",
                        "--method",
                        "GET",
                        f"repos/{REPO}/commits/{HEAD}/check-runs?per_page=100&page=1",
                    ],
                    cwd=RUN,
                    capture_output=True,
                    timeout=min(40, remaining),
                    check=False,
                )
                raw, errors, code = process.stdout, process.stderr, process.returncode
            except subprocess.TimeoutExpired as error:
                raw, errors, code = error.stdout or b"", error.stderr or b"", 124
            # Preserve exact response/error bytes before any JSON parsing.
            write_once(prefix.with_suffix(".stdout-private.bin"), raw)
            write_once(prefix.with_suffix(".stderr-private.bin"), errors)
            status, count = "http_or_cli_error", None
            if code == 0:
                try:
                    status, count = classify(json.loads(raw))
                except (ValueError, TypeError, AttributeError, KeyError):
                    status = "invalid_json_or_schema"
            observed = {
                "kind": "bounded_get_observation",
                "attempt": used,
                "finished_at_utc": utc(),
                "monotonic_elapsed_seconds": time.monotonic() - started,
                "return_code": code,
                "stdout_sha256": sha(raw),
                "stderr_sha256": sha(errors),
                "status": status,
                "github_actions_check_count": count,
                "head_sha": HEAD,
                "complete_window_proof": False,
                "mutations": 0,
            }
            write_once(prefix.with_suffix(".receipt.json"), encode(observed))
            print(json.dumps(observed), flush=True)
            if status == "all_exact_head_checks_green":
                write_once(OUT / "green-observed.json", encode(observed))
                break
            wait_until = min(began.timestamp() + INTERVAL_SECONDS, cutoff.timestamp())
            while time.time() < wait_until:
                if (OUT / "stop-observing").exists():
                    break
                time.sleep(min(1, max(0, wait_until - time.time())))
        print(
            json.dumps(
                {
                    "kind": "bounded_get_observer_finished",
                    "at_utc": utc(),
                    "GET_attempts_reserved": used,
                    "maximum_attempts": MAX_ATTEMPTS,
                    "cutoff_reached": datetime.now(UTC) >= cutoff,
                    "mutations": 0,
                }
            ),
            flush=True,
        )


if __name__ == "__main__":
    main()
